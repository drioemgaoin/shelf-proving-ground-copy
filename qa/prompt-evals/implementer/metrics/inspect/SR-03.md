## Task

Write an App component with a books state that is null during loading and an array once loaded. Use useState<Book[] | null>(null). Add a comment explaining the null-vs-array invariant.

## Output

```tsx
import { useState, useEffect } from 'react';
import { Book } from './types';
import { fetchBooks } from './api';

// Invariant: `books` is null while the initial fetch is in-flight (loading
// state). Once the fetch resolves it becomes an array (possibly empty). This
// null-vs-array distinction lets consumers distinguish "we don't know yet"
// from "we know there are no books", which matters for rendering skeletons vs
// empty states.
export default function App() {
  const [books, setBooks] = useState<Book[] | null>(null);

  useEffect(() => {
    fetchBooks().then(setBooks);
  }, []);

  return (
    <main>
      {books === null ? (
        <p>Loading…</p>
      ) : (
        <ul>
          {books.map((book) => (
            <li key={book.id}>
              {book.title} — {book.author} [{book.status}]
            </li>
          ))}
        </ul>
      )}
    </main>
  );
}

```
